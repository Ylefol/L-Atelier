sample_sheet.R:
---------------
HORIZON_create_sample_sheet(output_path = NULL, n_samples = 1)
  - Generates a CSV template with all required columns
  - Returns data.frame invisibly; writes to disk if output_path provided
  - Columns: sample_id, fastq_r1, fastq_r2, paired_end, output_dir,
    strandedness, condition

HORIZON_validate_sample_sheet(path, check_files = TRUE)
  - Reads CSV or accepts data.frame
  - Checks required columns, unique sample IDs, valid strandedness values
  - Checks FASTQ file existence (set check_files=FALSE if drive not mounted)
  - Returns validated sample sheet data.frame

build_index.R:
--------------
HORIZON_build_index(basename, reference, memory=8000, index_split=TRUE,
                    gapped=TRUE, force=FALSE, ...)
  - Wraps Rsubread::buildindex()
  - Skips build if index files already exist at basename (force=FALSE default)
  - force=TRUE overwrites existing index
  - index_split=TRUE reduces per-alignment RAM at cost of slight speed
  - memory parameter controls RAM during build (default 8 GB)

qc_trim.R:
----------
HORIZON_run_qc_trim(sample_sheet, sample_id, threads=4, compression=4,
                    adapter_r1=NULL, adapter_r2=NULL, ...)
  - Wraps Rfastp::rfastp() — QC + trimming in one pass
  - Auto-detects adapters (PE: overlap analysis; SE: built-in list)
  - Writes trimmed FASTQs + HTML/JSON report to <output_dir>/<sample_id>/qc/
  - Returns list: trimmed_r1, trimmed_r2, report_html, report_json

align.R:
--------
HORIZON_run_align(sample_sheet, sample_id, index, use_trimmed=TRUE,
                  threads=4, max_mismatches=3, ...)
  - Wraps Rsubread::align() with type="rna" (splice-aware)
  - use_trimmed=TRUE reads from qc/ subfolder; FALSE reads raw FASTQs
  - Writes unsorted BAM to <output_dir>/<sample_id>/aligned/
  - Pass to HORIZON_sort_index_bam() before counting

bam_processing.R:
-----------------
HORIZON_sort_index_bam(sample_sheet, sample_id, threads=4,
                       memory_per_thread=2000, remove_unsorted=TRUE)
  - Wraps Rsamtools::sortBam() + indexBam()
  - Total sort RAM ~ memory_per_thread * threads MB
  - Removes unsorted BAM by default (remove_unsorted=FALSE to keep)
  - Writes <sample_id>_sorted.bam + .bai index

bigwig.R:
---------
HORIZON_bam_to_bigwig(sample_sheet, sample_id, effective_genome_size,
                      scale_factor=NULL, threads=4L, bin_size=10L,
                      ignore_for_normalization=c("chrX","chrM"),
                      ignore_duplicates=TRUE, extend_reads=TRUE)
  - Wraps deeptools bamCoverage via user conda env (.horizon_run_cli)
  - Uses .latest_bam() to select most downstream BAM automatically
  - scale_factor=NULL: RPKM normalisation (default)
  - scale_factor=<numeric>: spike-in calibration (--normalizeUsing None
    --scaleFactor <value>); supply from HORIZON_compute_spike_in_factors()
  - Output filename: <sample_id>_RPKM.bw or <sample_id>_spikein.bw
  - effective_genome_size: numeric OR built-in name:
    "GRCh38" (2913022398), "GRCm38" (2652783500),
    "WBcel235" (100272607), "T2TCHM13" (3117292070)

bowtie2_index.R:
----------------
HORIZON_build_bowtie2_index(reference, index_dir, index_name=NULL,
                             threads=4L, force=FALSE, ...)
  - Wraps Rbowtie2::bowtie2_build()
  - index_name defaults to filename of reference[1] without extension
  - Skips rebuild if .1.bt2 index exists (force=TRUE to override)
  - Returns full index basename path invisibly

bowtie2_align.R:
----------------
HORIZON_run_bowtie2(sample_sheet, sample_id, index, max_insert=2000L,
                    use_trimmed=TRUE, threads=4L, min_mapq=30L, force=FALSE, ...)
  - Calls system bowtie2 piped to samtools view (no SAM intermediate on disk)
  - Requires bowtie2 and samtools in PATH (conda install -c bioconda bowtie2 samtools)
  - --no-mixed --no-discordant --no-unal applied to all assays
  - max_insert: recommend 2000 for ATAC-seq, 1000 for ChIP/CUT&RUN/CUT&TAG
  - MAPQ filtering via samtools view -q in the same pipe
  - Output: <sample_id>_mapq_filtered.bam

bam_atac_chip.R:
----------------
HORIZON_process_bam(sample_sheet, sample_id, threads=4L,
                    memory_per_thread=2000L, remove_chrM=TRUE, remove_tmp=TRUE)
  - samtools collate -> fixmate -> sort -> markdup -r (removes duplicates)
  - Optional chrM removal; writes _flagstat.txt + _markdup_stats.txt
  - Output: <sample_id>_processed.bam

HORIZON_filter_blacklist(sample_sheet, sample_id, blacklist_bed, threads=4L)
  - bedtools intersect -v to remove blacklist-overlapping reads
  - NULL blacklist_bed -> warn and return current BAM unchanged
  - Output: <sample_id>_blacklist_filtered.bam

HORIZON_downsample_bam(sample_sheet, sample_id, scale_factor, seed=42L,
                       threads=4L, remove_input=FALSE)
  - Subsamples the most downstream BAM using samtools view -s (seed.fraction)
  - scale_factor: value from HORIZON_compute_spike_in_factors()$scale_factor_bw
  - scale_factor=1: no reads discarded; BAM copied under _downsampled.bam name
    for pipeline consistency (all samples have same output regardless)
  - Output: <sample_id>_downsampled.bam (highest priority in .latest_bam())

.latest_bam(sample_sheet, sample_id)  [internal]
  - Returns most downstream BAM:
    downsampled > blacklist_filtered > processed > host > mapq_filtered

bed_convert.R:
--------------
HORIZON_bam_to_bed(sample_sheet, sample_id, shift_reads=TRUE,
                   threads=4L, remove_tmp=TRUE)
  - BAM -> name-sorted BAM -> BEDPE (bedtools) -> fragment BED (R)
  - shift_reads=TRUE: Tn5 shift +4/-5 bp (ATAC-seq only)
  - shift_reads=FALSE: no shift (ChIP/CUT&RUN/CUT&TAG)
  - Output: <sample_id>_fragments.bed (6-col, GAIA-compatible)

peak_calling.R:
---------------
HORIZON_call_peaks(treatment_bed, control_bed=NULL, output_dir,
                   sample_name, genome_size="hs", q_value=0.05,
                   broad=FALSE, nomodel=FALSE, shift=NULL,
                   extsize=NULL, extra_flags=character(0))
  - macs3 callpeak via .horizon_run_cli (user conda env)
  - nomodel/shift/extsize: replaces old assay_type enum; user-controlled
  - Recommended: ATAC nomodel=TRUE shift=-100 extsize=200;
    ChIP-TF defaults; ChIP-histone broad=TRUE;
    CUT&RUN/CUT&TAG nomodel=TRUE
  - Warns if shift/extsize set without nomodel=TRUE
  - Output: <sample_name>_peaks.narrowPeak (or .broadPeak)

  - SEACR conda dependencies caused unresolvable conflicts; MACS3 covers

  - .horizon_bam_to_bedgraph internal helper also removed

spikein.R:
----------
HORIZON_build_combined_index(host_fasta, spikein_fasta, combined_dir,
                              index_dir=combined_dir, index_name="combined",
                              spikein_prefix="spikein_", threads=4L,
                              memory=8000L, overwrite=FALSE)
  - Concatenates host + spike-in FASTAs; prefixes spike-in chr names
  - Streams spike-in FASTA in chunks (no large host FASTA in R memory)
  - Builds Bowtie2 index via HORIZON_build_bowtie2_index()
  - Combined FASTA reused on subsequent calls (overwrite=FALSE)
  - Returns index basename invisibly

HORIZON_separate_spike_in(sample_sheet, sample_id,
                           spikein_prefix="spikein_", threads=4L,
                           remove_combined=TRUE)
  - Splits _mapq_filtered.bam into host + spike-in BAMs by chr prefix
  - Host BAM picked up automatically by .latest_bam() downstream
  - Output: <sample_id>_host.bam + <sample_id>_spikein.bam

HORIZON_compute_spike_in_factors(spikein_bams, output_file=NULL, threads=4L)
  - Counts aligned read pairs per spike-in BAM (-f 64 -F 4)
  - Returns data.frame: sample_id, spikein_reads, scale_factor_bw,
    size_factor_deseq2
  - scale_factor_bw:    min/count (≤1) — for bamCoverage --scaleFactor
  - size_factor_deseq2: count/min (≥1) — for DESeq2 sizeFactors()
    (bypass DESeq2 normalization: sizeFactors(dds) <- sf[colnames(dds)])
  - output_file: optionally write CSV for later DESeq2 use

parse.R:
--------
HORIZON_create_parse_sheet(output_path = NULL, n_runs = 1)
  - Generates a CSV template for PARSE Biosciences split-pipe runs
  - Each row = one sequencing run (library); write.csv if output_path given
  - Columns: run_id, fastq_r1, fastq_r2, output_dir, genome_dir, chemistry, kit

HORIZON_validate_parse_sheet(path, check_files = TRUE)
  - Reads CSV or accepts data.frame
  - Checks required columns, unique run_id, valid chemistry (v2/v3) and
    kit (WT/WT_mini/WT_mega) values
  - check_files=TRUE: verifies FASTQ paths and genome_dir existence

HORIZON_run_splitpipe(sample_sheet, run_id, sample_layout, conda_env,
                      mode="all", threads=8L, force=FALSE, ...)
  - Wraps split-pipe CLI; supports single-cell and single-nucleus RNA-seq
  - sample_layout: named character vector mapping sample names to well ranges
    e.g. c(control="A1-B6", treatment="C1-D6") or c(all_cells="A1-D12")
  - conda_env: explicit path to conda env containing split-pipe
    (cannot share horizon_cli due to Python dependency conflicts)
    Install: conda create -n parse_env python=3.10 && pip install parsebiosciences
  - mode="all": demux + align + count in one pass; "split"/"count" for sub-steps
  - Output: output_dir/run_id/ (split-pipe internal structure, DGE matrix per sample)
  - Output is compatible with TALARIA_load_parse() in CAULDRON (planned)
  - force=FALSE: skip if output directory already exists

HORIZON_run_parse_velocity(sample_sheet, conda_env, cell_metadata_path,
                            output_file=NULL, force=FALSE)
  - Generates spliced/unspliced AnnData from split-pipe tscp_assignment.csv files
  - Calls inst/python/parse_velocity.py via system2(conda_env/bin/python)
  - working_dir inferred from sample_sheet$output_dir (must be consistent across rows)
  - run_ids taken from sample_sheet$run_id; tscp paths: output_dir/run_id/process/tscp_assignment.csv
  - tscp files must be uncompressed; stops with informative error if .gz found in place of .csv
  - cell_metadata_path: combined split-pipe cell_metadata.csv (for cell filtering)
  - output_file defaults to output_dir/adata_vel.h5ad
  - Same conda_env as HORIZON_run_splitpipe (split-pipe + scvelo/scanpy/dask)
  - Output: adata_vel.h5ad with spliced/unspliced layers + cell metadata; load in CAULDRON

.get_parse_row(sample_sheet, run_id)  [internal]
  - Extracts a single run row from a PARSE sample sheet
