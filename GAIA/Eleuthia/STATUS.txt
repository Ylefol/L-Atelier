sample_sheet.R:
---------------
ELEUTHIA_validate_sample_sheet(sample_sheet, check_files, check_beds, verbose)
  - Validate sample sheet structure and values
  - Checks required columns, format values, file existence
  - Generates unique sample_id for each row
  - omics column accepts any string (free-form label)

ELEUTHIA_get_omics_subset(sample_sheet, omics)
  - Extract samples of specific omics type
  - Accepts any omics string

ELEUTHIA_summarize_sample_sheet(sample_sheet)
  - Print summary of sample sheet structure
  - Cross-tabulation of omics x group

ELEUTHIA_load_quant_beds(sample_sheet, omics, verbose)
  - Load fragment BED files from bed_loc column
  - Returns named list of BED data.frames

peak_loading.R:
---------------
ELEUTHIA_load_narrowpeak(file_path, min_score, verbose)
  - Load narrowPeak format file
  - Optional score filtering

ELEUTHIA_load_peaks_from_sheet(sample_sheet, omics, min_score, verbose)
  - Load peaks for all samples of an omics type
  - Returns named list

ELEUTHIA_load_bed(file_path)
  - Load generic BED file (3-6 columns)

ELEUTHIA_load_bed_from_sheet(sample_sheet, omics, verbose)
  - Load BED files for all samples of an omics type

ELEUTHIA_create_consensus_peaks(peak_list, method, min_overlap, verbose)
  - Create consensus peak set from multiple samples
  - Methods: union, intersect, merge

ELEUTHIA_merge_fragments(bed_files, merge_distance, min_fragments, extend,
                          chrom_sizes, pool_samples, verbose)
  - Merge fragments into candidate regions with noise filter
  - min_fragments (default=10) removes obvious noise during merge
  - chrom_sizes: optional, clamps extended ends to chromosome boundaries
  - First step of two-step workflow for quantile-based selection

ELEUTHIA_select_regions(candidate_regions, min_fragments, quantile_threshold, verbose)
  - Filter candidate regions by threshold
  - Supports absolute threshold (min_fragments) or quantile-based (quantile_threshold)
  - If both provided, uses more stringent

ELEUTHIA_plot_region_distribution(candidate_regions, show_quantiles, log_scale, title)
  - Visualize fragment count distribution across candidate regions
  - Shows quantile thresholds to help choose selection criteria
  - Prints quantile table to console

ELEUTHIA_call_regions_from_fragments(bed_files, merge_distance, min_fragments,
                                       extend, chrom_sizes, pool_samples, verbose)
  - Convenience wrapper combining merge + select
  - For quick one-step workflow
  - chrom_sizes passed through to merge_fragments
  - Use two-step workflow for quantile-based selection

ELEUTHIA_peaks_to_saf(consensus_peaks)
  - Convert peaks to SAF format for featureCounts

ELEUTHIA_expand_regions(regions, window_size, chrom_sizes, verbose)
  - Symmetrically expand genomic regions by window_size bp
  - Clamps start to 0, optionally clamps end to chromosome size
  - Works with window_size = 0 (returns unchanged with metadata)
  - Adds original_width and window_size columns to output

ELEUTHIA_quantify_bed(sample_sheet, regions, omics, verbose)
  - Quantify fragments in regions from BED files
  - Memory-efficient: processes one file at a time
  - Returns quant_result object

stratification_report.R:
------------------------
ELEUTHIA_stratification_report(cluster_result, stability_result, char_result,
                                famd_result, comparison_result, qc_result,
                                output_dir, prefix, save_tables, save_plots, ...)
  - Comprehensive report for patient stratification pipeline
  - Accepts all pipeline result objects (cluster, stability, char, FAMD, comparison, QC)
  - Generates formatted text report with sections:
    - Data overview, Clustering results, Stability, Variable importance,
      FAMD summary, Variable refinement, Recommendations
  - Exports tables as CSV (cluster assignments, variable importance, etc.)
  - Generates and saves plots:
    - Silhouette plot (always)
    - K selection plot (when k="auto" was used)
    - Cluster Sankey diagram as HTML (when k="auto" was used)
    - FAMD plots (individuals, scree, variables) when famd_result provided
    - Radar plot when char_result provided
  - Returns report text, tables, and list of saved files

ELEUTHIA_quick_summary(cluster_result, stability_result, char_result)
  - Quick one-page console summary for interactive exploration
  - Shows clustering stats, stability, top 5 variables

discriminative_report.R:
------------------------
ELEUTHIA_discriminative_report(glasso_fit, cv_result, selected, final_model,
                                coef_table, encoded, output_dir, prefix, ...)
  - Comprehensive report for discriminative variable analysis pipeline
  - Accepts: Group LASSO fit, CV results, selected variables, final model, coefficients
  - Generates sections: Data overview, Variable selection, Final model, Coefficients
  - Exports tables: selected_variables.csv, coefficients.csv
  - Generates plots: forest_plot.png, lambda_path.png
  - Includes interpretation guide for odds ratios

ELEUTHIA_discriminative_quick_summary(glasso_fit, cv_result, selected)
  - Quick one-page console summary for interactive exploration
  - Shows data info, lambda selection, selected variables

wgcna_export.R:
---------------
ELEUTHIA_export_enrichment(enrichment, output_dir, prefix, by_source, by_module,
                            save_plots, plot_top_n, plot_format, plot_width,
                            plot_height_per_module, save_rds, verbose)
  - Export gprofiler2 enrichment results (from APOLLO_enrich_gost)
  - Creates: combined results CSV, summary CSV
  - Optional split by source (GO:BP, KEGG, etc.) in subdirectory
  - Optional split by module in subdirectory
  - save_plots: generates multi-module dotplot per source (uses AETHER_plot_gost_dotplot)
  - plot_format: "png", "pdf", or "both"
  - Optional RDS export for full gost_enrichment object
  - Standalone function for non-WGCNA enrichment use cases

ELEUTHIA_export_wgcna_results(output_dir, modules, trait_cor, gene_sig, hubs,
                               enrichment, power_result, cluster_result, wgcna_data,
                               save_plots, plot_format, prefix, save_rds, verbose)
  - Export WGCNA results to CSV files and plots
  - Creates: gene-module assignments, module summary, eigengenes,
    trait correlations, p-values, significant associations,
    gene significance, hub genes
  - Module-specific gene lists in subdirectory
  - Optional enrichment: exports to {prefix}_enrichment/ with dotplots
  - save_plots: generates plots to {prefix}_plots/:
    - power_selection.png (if power_result provided)
    - module_trait_heatmap.png (if trait_cor provided)
    - sample_dendrogram.png (if cluster_result provided)
  - plot_format: "png", "pdf", or "both"
  - Optional RDS export for R objects

ELEUTHIA_wgcna_report(output_dir, wgcna_data, power_result, modules, trait_cor,
                       gene_sig, hubs, prefix, verbose)
  - Generate formatted text report
  - Sections: Data summary, Power selection, Module detection,
    Trait correlations, Hub genes
  - Includes top associations and hub gene lists
